    drawCustomCursor() {
        if (!this.cursorImage || !this.cursorImage.complete) return;

        const cursorSize = 24; // Tamaño del cursor en píxeles
        
        // Dibujar la imagen del cursor en la posición del mouse
        // Centrado en el punto del mouse
        this.ctx.drawImage(
            this.cursorImage,
            this.mouse.x - cursorSize / 2,
            this.mouse.y - cursorSize / 2,
            cursorSize,
            cursorSize
        );
    }
